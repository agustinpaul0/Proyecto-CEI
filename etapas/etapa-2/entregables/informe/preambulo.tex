% ==============================================================================
% Notación de la gramática — la convención tipográfica de la cátedra
% ==============================================================================
% \nt{X}  no terminal: máquina de escribir, negro, como <X> en el PDF original
% \tl{x}  terminal:    sans-serif negrita, azul, como los terminales del PDF
%
% Los dos van en \mbox: un símbolo de la gramática no se parte nunca a mitad de
% renglón. Sin esto, LaTeX corta `while` en «whi-le» y `intLiteral` en
% «intLite-ral», que en una gramática es ilegible.
\newcommand{\nt}[1]{\mbox{\texttt{<#1>}}}
\newcommand{\tl}[1]{\mbox{\textsf{\bfseries\color{terminal}#1}}}
\newcommand{\eps}{\ensuremath{\epsilon}}
\newcommand{\deriva}{\mbox{\texttt{::=}}}
\newcommand{\alt}{\mbox{\ensuremath{\;\mid\;}}}
\newcommand{\prim}[1]{\textup{Primeros}(#1)}

% Bloque de producciones. Bandera a la izquierda (nunca justificado: justificar
% una secuencia de símbolos que no se pueden partir abre huecos enormes) y
% sangría francesa, para que la continuación de una producción larga se vea
% como continuación y no como una producción nueva.
\newenvironment{producciones}{%
  \par\smallskip\begingroup
  \setlength{\parindent}{0pt}%
  \setlength{\leftskip}{1.2em}%
  \setlength{\parskip}{0.45ex}%
  \raggedright
  \hyphenpenalty=10000 \exhyphenpenalty=10000 \hbadness=10000
  \everypar{\hangindent=2.2em\hangafter=1}%
  \obeylines}{\par\endgroup\smallskip}

\newcommand{\regla}[2]{\nt{#1} \deriva{} #2}
% Marca de logro: va al final y en línea. Con bandera a la izquierda un \hfill
% no empuja al margen, así que se pone pegada en vez de alineada a la derecha.
\newcommand{\marca}[2]{\quad{\footnotesize[#1\,\texttt{#2}]}}

% --- Cajas ---------------------------------------------------------------------
% Sin fondos: un filete al costado, que separa sin ensuciar la página.
\newenvironment{destacado}[2]{%
  \par\smallskip\begingroup
  \setlength{\leftskip}{1.1em}%
  \noindent\llap{\textcolor{tinta}{\vrule width 1.5pt height 1.6ex depth 0pt}\hspace{0.7em}}%
  {\bfseries\color{#1}#2}\par\nobreak\smallskip\ignorespaces
}{\par\endgroup\smallskip}

\newenvironment{conflicto}[1]{\begin{destacado}{tinta}{Conflicto LL(1) --- #1}}{\end{destacado}}
\newenvironment{ambiguedad}[1]{\begin{destacado}{tinta}{Ambigüedad --- #1}}{\end{destacado}}
\newenvironment{decision}[1]{\begin{destacado}{tinta}{Decisión --- #1}}{\end{destacado}}
